# Fair Bot v2 — Expectimax Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rewrite the fair bot search from alpha-beta (which cheats by reading the deck) to expectimax — branching on probability distributions at draw nodes so the bot has zero exposure to hidden information.

**Architecture:** Replace the `alphabeta` function and `pickFairBotAction` entry point. Keep the `OpponentTracker` class and eval factors unchanged. Add `combinations()` and `hypergeoProbAtLeast()` math helpers. Add `applyTakeWithChanceNodes()` which constructs states at draw nodes by branching on all possible card types weighted by probability. Sells and exchanges still use `applyAction` (safe — they never draw from deck). The eval function drops `realMarketIds` since all market cards are now explicitly constructed or original.

**Tech Stack:** TypeScript, vitest

## Key Insight: Which Actions Are Safe to Use `applyAction` For?

| Action | Draws from deck? | Safe for applyAction? |
|---|---|---|
| SELL | Never | Yes — safe at all depths |
| TAKE_EXCHANGE | Never — returned cards fill market slots | Yes — safe at all depths |
| TAKE_SINGLE | Yes — draws deck[0] | No — must construct states manually |
| TAKE_CAMELS | Yes — draws multiple from deck | No — must construct states manually |

This simplifies the implementation significantly: only TAKE_SINGLE and TAKE_CAMELS need chance nodes. Sells and exchanges can use the engine directly.

```typescript
export function combinations(n: number, k: number): number {
  if (k < 0 || k > n) return 0
  if (k === 0 || k === n) return 1
  let result = 1
  for (let i = 0; i < k; i++) {
    result = result * (n - i) / (i + 1)
  }
  return result
}

export function hypergeoProbAtLeast(available: number, total: number, draws: number, needed: number): number {
  if (needed <= 0) return 1
  if (available < needed || draws < needed) return 0
  let pFewer = 0
  for (let k = 0; k < needed; k++) {
    pFewer += combinations(available, k) * combinations(total - available, draws - k) / combinations(total, draws)
  }
  return 1 - pFewer
}
```

```typescript
let _nextFakeId = 9000

// Construct a state after TAKE_SINGLE without drawing from the deck.
// Returns the state with an empty market slot (card removed, added to player's hand).
// The caller is responsible for filling the slot via chance node branching.
function applyTakeSingleFair(state: GameState, marketIndex: number): GameState | null {
  const card = state.market[marketIndex]
  if (!card || card.type === 'camel') return null
  const player = state.players[state.activePlayer]
  if (player.hand.length >= 7) return null

  const newMarket = [...state.market]
  newMarket.splice(marketIndex, 1) // remove card, market now has 4

  const newPlayer = { ...player, hand: [...player.hand, card] }
  const players = [...state.players] as [typeof player, typeof player]
  players[state.activePlayer] = newPlayer

  return {
    ...state,
    market: newMarket,
    players,
    activePlayer: state.activePlayer === 0 ? 1 : 0 as 0 | 1,
  }
}

// Construct a state after TAKE_CAMELS without drawing from the deck.
// Returns the state with camels removed from market (fewer than 5 cards).
// The caller fills the empty slots via chance nodes.
function applyTakeCamelsFair(state: GameState): GameState | null {
  const camels = state.market.filter(c => c.type === 'camel')
  if (camels.length === 0) return null

  const newMarket = state.market.filter(c => c.type !== 'camel')
  const player = state.players[state.activePlayer]
  const newPlayer = { ...player, herd: player.herd + camels.length }
  const players = [...state.players] as [typeof player, typeof player]
  players[state.activePlayer] = newPlayer

  return {
    ...state,
    market: newMarket,
    players,
    activePlayer: state.activePlayer === 0 ? 1 : 0 as 0 | 1,
  }
}

// Fill empty market slots by branching on all possible card types.
// Returns the expected score across all possible draws.
function fillMarketChanceNode(
  state: GameState, // market has fewer than 5 cards
  deckComposition: Record<CardType, number>, // what's in the deck (by type, not order)
  deckSize: number,
  depth: number,
  myIndex: 0 | 1,
  tracker: OpponentTracker,
  deadline: number,
  maxChanceNodes: number,
): number {
  if (state.market.length >= 5 || deckSize <= 0) {
    // Market is full or deck is empty — check round end and continue search
    const finalState = checkRoundEndFair(state)
    return expectimax(finalState, depth, myIndex, tracker, deadline, maxChanceNodes)
  }

  // Branch on each possible card type for this slot
  let expectedScore = 0
  for (const type of ALL_TYPES) {
    if (deckComposition[type] <= 0) continue
    const prob = deckComposition[type] / deckSize

    // Construct state with this card type filling the next slot
    const fakeCard: Card = { id: _nextFakeId++, type }
    const newMarket = [...state.market, fakeCard]
    const newDeckComp = { ...deckComposition, [type]: deckComposition[type] - 1 }
    const newState = { ...state, market: newMarket }

    // Recurse to fill more slots if needed, or continue search
    expectedScore += prob * fillMarketChanceNode(
      newState, newDeckComp, deckSize - 1, depth, myIndex, tracker, deadline, maxChanceNodes - 1,
    )
  }

  return expectedScore
}

// Check if round should end (3 piles depleted or market < 5 with no deck)
function checkRoundEndFair(state: GameState): GameState {
  if (state.market.length < 5) {
    // In our construction, deck is implicit — if we couldn't fill market, round ends
    return { ...state, phase: 'round-end' as const }
  }
  const depleted = GOOD_ORDER.filter(g => state.tokens[g].length === 0).length
  if (depleted >= 3) {
    return { ...state, phase: 'round-end' as const }
  }
  return state
}

// Compute deck composition: unaccounted cards minus opponent's unknown hand
function computeDeckComposition(tracker: OpponentTracker, unaccounted: Record<CardType, number>): Record<CardType, number> {
  const deck: Record<CardType, number> = { ...unaccounted }
  // Remove the unknown opponent hand cards from the deck pool
  // We don't know which types they are, but we can't just subtract expected values.
  // For chance node branching, we use the full unaccounted pool and accept that
  // some of those cards might be in the opponent's hand.
  // This is the honest approximation — the bot doesn't know where unknowns are.
  // Camels are special: they're always in the deck (never in hands).
  // For non-camel types, reduce proportionally by unknownInHand.
  if (tracker.unknownInHand > 0) {
    const totalNonCamel = Object.values(deck).reduce((s, n) => s + n, 0) - deck.camel
    if (totalNonCamel > 0) {
      const handFraction = tracker.unknownInHand / (totalNonCamel + tracker.unknownInHand)
      for (const good of GOOD_ORDER) {
        deck[good] = Math.max(0, deck[good] * (1 - handFraction))
      }
    }
  }
  return deck
}
```

```typescript
function generateOpponentActions(state: GameState, tracker: OpponentTracker, unaccounted: Record<CardType, number>): { action: Action; weight: number }[] {
  const oppIndex = state.activePlayer
  const opp = state.players[oppIndex]
  const actions: { action: Action; weight: number }[] = []
  const totalUnaccounted = Object.values(unaccounted).reduce((s, n) => s + n, 0)

  // Takes from market — always concrete (market is visible)
  if (opp.hand.length < 7) {
    for (let i = 0; i < state.market.length; i++) {
      if (state.market[i].type !== 'camel') {
        actions.push({ action: { type: 'TAKE_SINGLE', marketIndex: i }, weight: 1 })
      }
    }
  }

  // Take camels
  if (state.market.some(c => c.type === 'camel')) {
    actions.push({ action: { type: 'TAKE_CAMELS' }, weight: 1 })
  }

  // Sells — known cards: concrete
  const knownCounts = new Map<Good, number>()
  for (const card of tracker.knownInHand) {
    if (card.type === 'camel') continue
    const g = card.type as Good
    knownCounts.set(g, (knownCounts.get(g) ?? 0) + 1)
  }
  for (const [good, count] of knownCounts) {
    const minQty = PRECIOUS.has(good) ? 2 : 1
    for (let qty = minQty; qty <= count; qty++) {
      actions.push({ action: { type: 'SELL', good, quantity: qty }, weight: 1 })
    }
  }

  // Sells — unknown cards: hypergeometric probability
  if (tracker.unknownInHand > 0) {
    for (const good of GOOD_ORDER) {
      const known = knownCounts.get(good) ?? 0
      const minSell = MIN_SELL[good]
      if (known >= minSell) continue // already covered by known sells above
      const needed = minSell - known
      const available = unaccounted[good] ?? 0
      const prob = hypergeoProbAtLeast(available, totalUnaccounted, tracker.unknownInHand, needed)
      if (prob > 0.1) {
        const qty = known + needed // sell exactly enough
        actions.push({ action: { type: 'SELL', good, quantity: qty }, weight: prob })
      }
    }
  }

  // Exchanges — only with known cards + camels (skip unknown card exchanges)
  // Use getProfitableExchanges only if opponent hand is fully known
  if (tracker.unknownInHand === 0) {
    const exchanges = getProfitableExchanges(state)
    for (const ex of exchanges) {
      actions.push({ action: ex, weight: 1 })
    }
  }

  return actions
}

function expectimax(
  state: GameState,
  depth: number,
  myIndex: 0 | 1,
  tracker: OpponentTracker,
  deadline: number,
  maxChanceNodes: number,
): number {
  if (state.phase !== 'playing' || depth <= 0 || Date.now() >= deadline) {
    return evalPosition(state, myIndex, tracker)
  }

  const me = state.players[myIndex]
  const opp = state.players[myIndex === 0 ? 1 : 0]
  const unaccounted = tracker.computeUnaccounted(me.hand, me.herd, state.market, state.discard, opp.herd)

  if (state.activePlayer === myIndex) {
    // Maximizer — bot's turn
    const rawActions = [...getLegalActions(state), ...getProfitableExchanges(state)]
    if (rawActions.length === 0) return evalPosition(state, myIndex, tracker)
    const actions = orderActions(rawActions, state, myIndex)

    let best = -Infinity
    for (const action of actions) {
      if (Date.now() >= deadline) break
      const s = evalAction(state, action, depth, myIndex, tracker, unaccounted, deadline, maxChanceNodes)
      if (s > best) best = s
    }
    return best === -Infinity ? evalPosition(state, myIndex, tracker) : best
  } else {
    // Minimizer — opponent's turn
    const oppActions = generateOpponentActions(state, tracker, unaccounted)
    if (oppActions.length === 0) return evalPosition(state, myIndex, tracker)

    // Weighted minimum: for weighted actions, compute weighted average of the worst outcomes
    let totalWeight = 0
    let weightedScore = 0
    const concreteActions: { action: Action; score: number }[] = []

    for (const { action, weight } of oppActions) {
      if (Date.now() >= deadline) break
      const s = evalAction(state, action, depth, myIndex, tracker, unaccounted, deadline, maxChanceNodes)
      if (weight >= 1) {
        // Concrete action — opponent will choose the worst for us
        concreteActions.push({ action, score: s })
      } else {
        // Probabilistic action — weighted contribution
        totalWeight += weight
        weightedScore += weight * s
      }
    }

    // The opponent picks the minimum across concrete actions and probabilistic actions
    let minConcrete = Infinity
    for (const { score } of concreteActions) {
      if (score < minConcrete) minConcrete = score
    }

    if (concreteActions.length === 0 && totalWeight > 0) {
      return weightedScore / totalWeight
    }
    if (totalWeight > 0) {
      const avgProbabilistic = weightedScore / totalWeight
      return Math.min(minConcrete, avgProbabilistic)
    }
    return minConcrete === Infinity ? evalPosition(state, myIndex, tracker) : minConcrete
  }
}

function evalAction(
  state: GameState,
  action: Action,
  depth: number,
  myIndex: 0 | 1,
  tracker: OpponentTracker,
  unaccounted: Record<CardType, number>,
  deadline: number,
  maxChanceNodes: number,
): number {
  if (action.type === 'SELL') {
    // Sells don't draw from deck — safe to use applyAction
    const result = applyAction(state, action)
    if (!result.ok) return -Infinity
    return expectimax(result.value, depth - 1, myIndex, tracker, deadline, maxChanceNodes)
  }

  if (action.type === 'TAKE_EXCHANGE') {
    // Exchanges don't draw from deck — returned cards fill market slots
    const result = applyAction(state, action)
    if (!result.ok) return -Infinity
    return expectimax(result.value, depth - 1, myIndex, tracker, deadline, maxChanceNodes)
  }

  if (maxChanceNodes <= 0) {
    // No more chance node budget — evaluate immediately
    return evalPosition(state, myIndex, tracker)
  }

  if (action.type === 'TAKE_SINGLE') {
    const partialState = applyTakeSingleFair(state, action.marketIndex)
    if (!partialState) return -Infinity
    const deckComp = computeDeckComposition(tracker, unaccounted)
    const deckSize = Object.values(deckComp).reduce((s, n) => s + n, 0)
    return fillMarketChanceNode(partialState, deckComp, deckSize, depth - 1, myIndex, tracker, deadline, maxChanceNodes - 1)
  }

  if (action.type === 'TAKE_CAMELS') {
    const partialState = applyTakeCamelsFair(state)
    if (!partialState) return -Infinity
    const deckComp = computeDeckComposition(tracker, unaccounted)
    const deckSize = Object.values(deckComp).reduce((s, n) => s + n, 0)
    return fillMarketChanceNode(partialState, deckComp, deckSize, depth - 1, myIndex, tracker, deadline, maxChanceNodes - 1)
  }

  return evalPosition(state, myIndex, tracker)
}
```

```typescript
export function pickFairBotAction(state: GameState, tracker: OpponentTracker): Action | null {
  if (state.phase !== 'playing') return null
  const myIndex = state.activePlayer

  const rootActions = orderActions(
    [...getLegalActions(state), ...getAllProfitableExchanges(state)],
    state,
    myIndex,
  )
  if (rootActions.length === 0) return null
  if (rootActions.length === 1) return rootActions[0]

  const isEndg = isEndgame(state)
  const deadline = Date.now() + (isEndg ? 12000 : 7000)
  const maxChanceNodes = 3 // limit chance node branching per path

  const me = state.players[myIndex]
  const opp = state.players[myIndex === 0 ? 1 : 0]
  const unaccounted = tracker.computeUnaccounted(me.hand, me.herd, state.market, state.discard, opp.herd)

  let bestAction = rootActions[0]
  let bestScore = -Infinity

  // Iterative deepening
  const depthCap = isEndg ? 20 : 8

  for (let depth = 1; depth <= depthCap; depth++) {
    if (Date.now() >= deadline) break

    let depthBest = rootActions[0]
    let depthBestScore = -Infinity
    let completedDepth = true

    for (const action of rootActions) {
      if (Date.now() >= deadline) { completedDepth = false; break }
      const s = evalAction(state, action, depth, myIndex, tracker, unaccounted, deadline, maxChanceNodes)
      if (s > depthBestScore) {
        depthBestScore = s
        depthBest = action
      }
    }

    if (completedDepth) {
      bestAction = depthBest
      bestScore = depthBestScore
    }
  }

  return bestAction
}
```
